1. Đổi resolution lúc đang chạy

Vì đây là một software rasterizer, mọi pixel đều do CPU tính, nghĩa là số pixel chính là thứ quyết định nó nhanh hay chậm. Cho người đọc tự kéo độ resolution xuống hay lên là cách tốt nhất để họ thấy điều đó.

Từ blog 1 tới giờ cái canvas luôn luôn là 400x300, kích thước được viết thẳng trong thẻ <canvas> và Rasterizer cấp phát buffer đúng một lần trong constructor.

	canvasRenderHeight = readResolution(canvasResHeightInput, 16, 1440, canvasRenderHeight);
	canvasResWidthInput.value = canvasRenderWidth.toString();
	canvasResHeightInput.value = canvasRenderHeight.toString();
	
	gameCanvas.width = canvasRenderWidth;

	projectionMatrix = Mat4x4.Projection(fov, gameCanvas.height, gameCanvas.width, znear, zfar);

Hàm readResolution sẽ đọc từ input và check NaN và min max: 
function readResolution(input, min, max, current) {
	let value = parseInt(input.value);
	if (isNaN(value)) return current;
	return Math.max(min, Math.min(max, value));
}
Còn hàm updateRenderResolution là hàm thật sự thay đổi resolution render cho engine, đây là những thứ cần phải làm:
  1. Tính lại projection matrix (nó có chứa tỉ lệ h / w, xem lại blog 3)
  2. Trong rasterizer: init lại imageData, screenBuffer, screenBuffer32, depthBuffer
Như vậy nó sẽ như sau: 
function updateRenderResolution() {
	gameCanvas.width = canvasRenderWidth;
	gameCanvas.height = canvasRenderHeight;

	projectionMatrix = Mat4x4.Projection(fov, gameCanvas.height, gameCanvas.width, znear, zfar);
	rasterizer.updateResolution(gameCanvas.width, gameCanvas.height);
}
Hàm Rasterizer.updateResolution: 
updateResolution(canvasWidth, canvasHeight) {
	this.canvasWidth = canvasWidth;
	this.canvasHeight = canvasHeight;

	this.imageData = this.ctx.createImageData(this.canvasWidth, this.canvasHeight);
	this.screenBuffer = this.imageData.data;
	this.screenBuffer32 = new Uint32Array(this.imageData.data.buffer);
	this.depthBuffer = new Float64Array(this.canvasWidth * this.canvasHeight);
	
	this.clearScreen();
}

Ta cần trả lời câu hỏi sau: Sao để chuyển từ 1 pixel ra 1 ký tự? Output từ 1 pixel ra 1 ký tự thì sẽ không giữ được thông tin màu (ký tự thuần thì không có màu). Tuy nhiên ta có thể dựa vào grayScale của màu để suy ra độ đậm nhạt, từ đó suy ra kí tự.

Đầu tiên ta thêm hàm tính grayScale, nó chỉ là trung bình của 3 channel, chia 255 để đưa nó về khoảng [0; 1]: 
grayScale01() {
	return (this.r + this.g + this.b) / 3.0 / 255.0;
}
Về suy ra kí tự, ta dùng dãy bên dưới để nội suy ra kí tự từ grayScale. 
let grayScaleASCII = "$@B%8&WM#*oahkbdpqwmZO0QLCJUYXzcvunxrjft/\|()1{}[]?-_+~<>i!lI;:,\"^`'. ";
Nguồn: Paul Bourke
Đầu dãy là $, @, B - mấy kí tự có mấy nét gần nhau, nhìn đậm. Cuối dãy là . và dấu cách, nhìn nhạt. Nên pixel càng "sáng" thì càng lấy kí tự ở cuối, tức là phải lấy 1 - grayScale01.

Về việc render thì ta phải sửa class Rasterizer, nhưng thật ra ta chỉ cần thêm reference tới asciiParagraph và if trong hàm drawCall: 
class Rasterizer {
	constructor(canvasWidth, canvasHeight, ctx, asciiParagraph) {
		...
		this.asciiParagraph = asciiParagraph;
		...
	}
	...
	drawCall(useASCII = false) {
		if (!useASCII) {
			this.ctx.putImageData(this.imageData, 0, 0);
		} else {
			let renderString = "";		// string bình thường là đủ
			let grayScaleASCII = "$@B%8&WM#*oahkbdpqwmZO0QLCJUYXzcvunxrjft/\\|()1{}[]?-_+~<>i!lI;:,\"^`'. ";
			for (let y = 0; y < this.canvasHeight; y++) {
				for (let x = 0; x < this.canvasWidth; x++) {
					let index = (y * this.canvasWidth + x) * 4;
					let color = new Color(this.imageData.data[index], this.imageData.data[index + 1], this.imageData.data[index + 2]);
					let grayScale01 = color.grayScale01();
					// nội suy với độ dài dãy
					let pixelLetter = grayScaleASCII[Math.floor((1 - grayScale01) * (grayScaleASCII.length - 1))];
					renderString += pixelLetter;
				}
				renderString += "\n";
			}

			if (this.asciiParagraph != null && this.asciiParagraph.innerText != renderString)
				this.asciiParagraph.innerText = renderString;
		}
	}
}
Cái this.asciiParagraph.innerText != renderString để cho nếu khung hình không đổi thì đừng cập nhật, paragraph mà bị cập nhật liên tục thì bạn sẽ không thể bôi đậm và copy, mất tính "magic".

async function requestMouseLook() {

const keyButtons = mobileControls.querySelectorAll("button[data-key]");
keyButtons.forEach((button) => {
	const key = (button.getAttribute("data-key"));		// sử dụng data-key ở trên
	button.addEventListener("pointerdown", (e) => {
		e.preventDefault();
		keyStates[key] = true;
		button.setPointerCapture(e.pointerId);	// xử lí tình huống nhấn nút rồi trượt ngón tay ra ngoài
	});
	const release = () => {
		keyStates[key] = false;
	};
	button.addEventListener("pointerup", release);
	button.addEventListener("pointercancel", release);	// trường hợp trình duyệt cướp con trỏ giữa chừng (vuốt, ...)
});
Mình dùng pointer event chứ không phải mousedown/touchstart, vì pointer event gộp chung chuột, cảm ứng và bút thành một API.

Phần nhìn xung quanh: Với chuột, mình có sẵn movementX và movementY của chuột trong event mousemove. Còn với nút bấm, mình tạo một cái object lookInput để cộng dồn vào: 
let lookInput = { x: 0, y: 0 };
let lookSpeed = 450;
...
const lookButtons = mobileControls.querySelectorAll("button[data-look]");
lookButtons.forEach((button) => {
	if (!button.dataset.look) return;
	const dir = button.dataset.look.split(",");
	const x = parseInt(dir[0]),
		y = parseInt(dir[1]);
	let pressed = false;
	button.addEventListener("pointerdown", (e) => {
		e.preventDefault();
		if (pressed) return;
		pressed = true;
		lookInput.x += x;
		lookInput.y += y;
		button.setPointerCapture(e.pointerId);
	});
	const release = () => {
		if (!pressed) return;
		pressed = false;
		lookInput.x -= x;	// trừ lại, thay vì set = 0, để khi 2 nút ngược nhau thì chúng triệt tiêu nhau
		lookInput.y -= y;
	};
	button.addEventListener("pointerup", release);
	button.addEventListener("pointercancel", release);
});
